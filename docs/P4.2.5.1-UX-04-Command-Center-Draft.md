## 三、Service Contract

### enterprise-command-center.service.ts

```typescript
import { useAuthStore } from '@/stores/auth'

/**
 * Command Center Service
 * 
 * 责任：从 Backend API 获取数据，包装为 Envelope
 * 禁止：缓存 / 状态管理 / UI 逻辑
 */

const API_BASE = '/api/enterprise'

async function fetchEnvelope<T>(
  path: string,
  source: string
): Promise<EnterpriseDataEnvelope<T>> {
  const authStore = useAuthStore()
  const headers: Record<string, string> = {}
  const token = authStore.getToken()
  if (token) headers['Authorization'] = `Bearer ${token}`

  const res = await fetch(`${API_BASE}/${authStore.tenantId}${path}`, { headers })
  const json = await res.json()

  // 包装 Envelope
  return {
    data: json.data as T,
    source,
    freshness: new Date().toISOString(),
    confidence: json.confidence,
    timestamp: new Date().toISOString(),
  }
}

/** 获取 Today Intelligence 指标 */
export function getTodayMetrics(): Promise<EnterpriseDataEnvelope<MetricValue>[]> {
  return fetchEnvelope('/dashboard/today-metrics', 'OperationEvent')
}

/** 获取 Top N 决策 */
export function getDecisionQueue(limit = 3): Promise<EnterpriseDataEnvelope<Decision>[]> {
  return fetchEnvelope(`/decisions/top?limit=${limit}`, 'DecisionEngine')
}

/** 获取执行中的 Actions */
export function getActiveActions(): Promise<EnterpriseDataEnvelope<ActionProgress>[]> {
  return fetchEnvelope('/actions/active', 'ActionLifecycle')
}

/** 获取待审批的 Actions */
export function getPendingApprovalActions(): Promise<EnterpriseDataEnvelope<ActionProgress>[]> {
  return fetchEnvelope('/actions/pending-approval', 'ActionLifecycle')
}

/** 获取 Business Signals */
export function getSignals(): Promise<EnterpriseDataEnvelope<Signal>[]> {
  return fetchEnvelope('/signals', 'OperationEvent')
}

/** 获取渠道健康 */
export function getChannelStatus(): Promise<EnterpriseDataEnvelope<ChannelAccount>[]> {
  return fetchEnvelope('/channels/status', 'ChannelSync')
}

/** 获取同步状态 */
export function getSyncStatus(): Promise<EnterpriseDataEnvelope<SyncStatus>> {
  return fetchEnvelope('/channels/sync-status', 'ChannelSync')
}

/** 获取最新 Outcome */
export function getLatestOutcome(): Promise<EnterpriseDataEnvelope<Outcome>> {
  return fetchEnvelope('/outcomes/latest', 'OutcomeIntelligence')
}

/** 证据链 */
export function getEvidence(decisionId: string): Promise<EnterpriseDataEnvelope<EvidenceNode[]>> {
  return fetchEnvelope(`/decisions/${decisionId}/evidence`, 'EvidenceGraph')
}

/** 决策操作 */
export function acceptDecision(decisionId: string, note: string) {
  return fetchEnvelope(`/decisions/${decisionId}/accept`, 'DecisionEngine')
}

export function rejectDecision(decisionId: string) {
  return fetchEnvelope(`/decisions/${decisionId}/reject`, 'DecisionEngine')
}
```

## 四、Data Envelope Mapping

### EnterpriseDataEnvelope v1.1（CTO 冻结）

```typescript
interface EnterpriseDataEnvelope<T> {
  data: T
  source: string
  freshness: string        // ISO 8601
  confidence?: number      // 0-100
  timestamp: string        // ISO 8601
}
```

### 每个 Section 的 Envelope 映射

| Section | data 结构 | source |
|---------|----------|--------|
| TodayIntelligence.events | `{ value: number, trend: string }` | `OperationEvent` |
| TodayIntelligence.decisions | `{ value: number, trend: string }` | `DecisionEngine` |
| TodayIntelligence.actions | `{ value: number, trend: string }` | `ActionLifecycle` |
| TodayIntelligence.outcomeDelta | `{ value: string, trend: string }` | `OutcomeIntelligence` |
| DecisionQueue.items | `Decision` | `DecisionEngine` |
| Execution.active | `ActionProgress` | `ActionLifecycle` |
| Execution.pendingApproval | `ActionProgress` | `ActionLifecycle` |
| Signals.items | `Signal` | `OperationEvent` |
| Signals.evidence | `EvidenceNode[]` | `EvidenceGraph` |
| Channels.accounts | `ChannelAccount` | `ChannelSync` |
| Channels.syncStatus | `SyncStatus` | `ChannelSync` |
| Outcomes.latest | `Outcome` | `OutcomeIntelligence` |

```
┌─────────────────────────────────────────────────────────┐
```

### 规则

| 状态 | 显示 | 不显示 |
|------|------|--------|
| Loading | Skeleton (per section) | Spinner, "Loading..." 文字 |
| Empty | EmptyState (icon + title + desc) | "No data" 文字, 空 div |
| Error | StatusBadge + Retry | Error 堆叠, Alert 弹窗 |
| Partial | 可用 Data + Error Section | 整页 Error |
