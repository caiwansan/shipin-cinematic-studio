## 五、Enterprise Context Composable

### 5.1 useEnterpriseContext()

```typescript
// composables/enterprise/useEnterpriseContext.ts

export interface EnterpriseContext {
  tenant: {
    id: string
    name: string
    source: string
    freshness: string
  }
  user: {
    id: string
    name: string
    role: 'ceo' | 'manager' | 'operator' | 'admin'
  }
  capabilities: string[]
  can: (capability: string) => boolean
}

export const useEnterpriseContext = (): EnterpriseContext => {
  const authStore = useAuthStore()
  
  return {
    tenant: {
      id: authStore.tenantId,
      name: authStore.tenantName,
      source: 'auth_store',
      freshness: 'login_time'
    },
    user: {
      id: authStore.userId,
      name: authStore.userName,
      role: authStore.userRole
    },
    capabilities: authStore.capabilities || [],
    can: (capability: string) => authStore.capabilities?.includes(capability) ?? false
  }
}
```
